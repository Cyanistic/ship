+// Saved native inputs only: independent strategy/size processes let the caller bound
+// quadratic ordered alignment without losing completed sizes or transitions.
+pub(super) fn run_saved(size: &str, strategy: &str) -> Result<(), Box<dyn std::error::Error>> {
+    assert!(!cfg!(debug_assertions), "run this benchmark with --release");
+    if !matches!(size, "80x24" | "120x40" | "160x50" | "80x50" | "160x25")
+        || !matches!(strategy, "json" | "default" | "ordered" | "ordered-probe")
+    {
+        return Err("expected approved size and json/default/ordered/ordered-probe".into());
+    }
+    let input = std::path::Path::new("size-results")
+        .join(size)
+        .join("captures.json");
+    let captures: Vec<(String, Screen)> = serde_json::from_slice(&std::fs::read(&input)?)?;
+    assert_eq!(captures.len(), 4);
+    assert_eq!(
+        captures
+            .iter()
+            .map(|(label, _)| label.as_str())
+            .collect::<Vec<_>>(),
+        [
+            "initial-filled",
+            "small-edit",
+            "one-row-scroll",
+            "unchanged"
+        ]
+    );
+    for (_, screen) in &captures {
+        assert_eq!(format!("{}x{}", screen.cols, screen.rows), size);
+        assert_eq!(
+            screen.cells.len(),
+            screen.cols as usize * screen.rows as usize
+        );
+    }
+    let output = std::path::Path::new("structdiff-size-results")
+        .join(size)
+        .join(strategy);
+    std::fs::create_dir_all(&output)?;
+    if strategy == "ordered-probe" {
+        let old = OrderedScreen::from(&captures[0].1);
+        let new = OrderedScreen::from(&captures[1].1);
+        let start = Instant::now();
+        let patch = black_box(&old).diff(black_box(&new));
+        let elapsed = start.elapsed().as_secs_f64();
+        let bytes = serde_json::to_vec(&patch)?;
+        let decoded = serde_json::from_slice(&bytes)?;
+        assert_eq!(old.apply(decoded), new);
+        std::fs::write(
+            output.join("probe.json"),
+            serde_json::to_vec_pretty(&json!({
+                "size": size, "transition": "small-edit", "single_diff_seconds": elapsed,
+                "patch_bytes": bytes.len(), "equality": "PASS", "samples": 1,
+                "warmup": 0, "not_a_percentile_measurement": true
+            }))?,
+        )?;
+        println!("ORDERED SINGLE DIFF PASS {size}: {elapsed:.6}s");
+        return Ok(());
+    }
+    let (warmup, samples) = if strategy == "ordered" {
+        (3, 10)
+    } else {
+        (20, 100)
+    };
+    let mut results = Results(Vec::new(), warmup, samples, true);
+    let initial = serde_json::to_vec(&captures[0].1)?;
+    assert_eq!(serde_json::from_slice::<Screen>(&initial)?, captures[0].1);
+    let mut client: Value = serde_json::from_slice(&initial)?;
+    let mut prior = serde_json::to_value(&captures[0].1)?;
+    let mut evidence = vec![
+        json!({"initial_snapshot_bytes": initial.len(), "equality": "PASS",
+        "input": input, "strategy": strategy, "warmup": warmup, "samples": samples}),
+    ];
+    for pair in captures.windows(2) {
+        let (label, new) = &pair[1];
+        let old = &pair[0].1;
+        evidence.push(match strategy {
+            "json" => json_transition(&mut results, label, old, new, &mut prior, &mut client),
+            "default" => struct_transition(&mut results, label, "structdiff-default", old, new),
+            "ordered" => {
+                let old_ordered = OrderedScreen::from(old);
+                let new_ordered = OrderedScreen::from(new);
+                assert_eq!(serde_json::to_vec(&old_ordered)?, serde_json::to_vec(old)?);
+                assert_eq!(serde_json::to_vec(&new_ordered)?, serde_json::to_vec(new)?);
+                struct_transition(
+                    &mut results,
+                    label,
+                    "structdiff-ordered",
+                    &old_ordered,
+                    &new_ordered,
+                )
+            }
+            _ => unreachable!(),
+        });
+        // Save each completed transition so a bounded process stop retains partial evidence.
+        std::fs::write(
+            output.join("measurements.json"),
+            serde_json::to_vec_pretty(&results.0)?,
+        )?;
+        std::fs::write(
+            output.join("wire-evidence.json"),
+            serde_json::to_vec_pretty(&evidence)?,
+        )?;
+    }
+    if strategy == "json" {
+        assert_eq!(
+            serde_json::from_value::<Screen>(client)?,
+            captures.last().unwrap().1
+        );
+    }
+    assert_eq!(results.0.len(), if strategy == "json" { 18 } else { 9 });
+    println!(
+        "SAVED BENCH PASS {size} {strategy}: {} metrics warmup={warmup} samples={samples}",
+        results.0.len()
     );
+    Ok(())
 }
